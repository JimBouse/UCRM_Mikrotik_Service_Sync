/system script remove [find name=update_pcq];
/system script add dont-require-permissions=no name=update_pcq source="# =============\
    ===============================\
    \n# Script: update_pcq\
    \n# Purpose: Create / maintain PCQ speed plans\
    \n#          based on address-lists named\
    \n#          Speed_Package-XX/YY\
    \n# ============================================\
    \n# NAT-friendly (connection-mark based)\
    \n# Separate UP / DOWN Simple Queues\
    \n# Robust cleanup using queue name parsing\
    \n# ============================================\
    \n\
    \n:local logPrefix \"update_pcq: \"\
    \n\
    \n:log info (\$logPrefix . \"Starting...\")\
    \n\
    \n# ---------- Helper: extract download/upload from list name ----------\
    \n:local parseSpeeds do={\
    \n    :local name \$1\
    \n    :local prefix \"Speed_Package-\"\
    \n    :if ([:pick \$name 0 [:len \$prefix]] != \$prefix) do={ :return \"\" }\
    \n\
    \n    :local speeds [:pick \$name [:len \$prefix] [:len \$name]]\
    \n    :local slashPos [:find \$speeds \"/\"]\
    \n    :if (\$slashPos = nil) do={ :return \"\" }\
    \n\
    \n    :local dl [:pick \$speeds 0 \$slashPos]\
    \n    :local ul [:pick \$speeds (\$slashPos + 1) [:len \$speeds]]\
    \n\
    \n    :if ([:tonum \$dl] = nil || [:tonum \$ul] = nil) do={ :return \"\" }\
    \n\
    \n    :return (\$dl . \"/\" . \$ul)\
    \n}\
    \n\
    \n# ---------- Collect unique active packages ----------\
    \n:local activePackages ({})\
    \n:foreach entry in=[/ip firewall address-list find where list~\"^Speed_Package-\"] do={\
    \n    :local listName [/ip firewall address-list get \$entry list]\
    \n    :local speeds [\$parseSpeeds \$listName]\
    \n    :if (\$speeds != \"\") do={\
    \n        :set (\$activePackages->\$speeds) \$listName\
    \n    }\
    \n}\
    \n\
    \n:log info (\$logPrefix . \"Found \" . [:len \$activePackages] . \" active speed packages\")\
    \n\
    \n# ---------- Create missing infrastructure ----------\
    \n:foreach speeds,listName in=\$activePackages do={\
    \n\
    \n    :local slashPos [:find \$speeds \"/\"]\
    \n    :local dl [:pick \$speeds 0 \$slashPos]\
    \n    :local ul [:pick \$speeds (\$slashPos + 1) [:len \$speeds]]\
    \n\
    \n    :local connMark    (\"sp-\" . \$dl . \"-\" . \$ul)\
    \n    :local pktDown     (\$connMark . \"-down\")\
    \n    :local pktUp       (\$connMark . \"-up\")\
    \n    :local pcqDown     (\"pcq-\" . \$connMark . \"-down\")\
    \n    :local pcqUp       (\"pcq-\" . \$connMark . \"-up\")\
    \n    :local simpleUp    (\"SP-\" . \$dl . \"-\" . \$ul . \"-UP\")\
    \n    :local simpleDown  (\"SP-\" . \$dl . \"-\" . \$ul . \"-DOWN\")\
    \n\
    \n    :log info (\$logPrefix . \"Processing \" . \$listName)\
    \n\
    \n    # 1. PCQ Queue Types\
    \n    :if ([:len [/queue type find where name=\$pcqDown]] = 0) do={\
    \n        /queue type add name=\$pcqDown kind=pcq \\\
    \n            pcq-rate=(\$dl . \"M\") pcq-classifier=dst-address \\\
    \n            pcq-limit=50KiB pcq-total-limit=2000KiB\
    \n        :log info (\$logPrefix . \"Created \" . \$pcqDown)\
    \n    }\
    \n    :if ([:len [/queue type find where name=\$pcqUp]] = 0) do={\
    \n        /queue type add name=\$pcqUp kind=pcq \\\
    \n            pcq-rate=(\$ul . \"M\") pcq-classifier=src-address \\\
    \n            pcq-limit=50KiB pcq-total-limit=2000KiB\
    \n        :log info (\$logPrefix . \"Created \" . \$pcqUp)\
    \n    }\
    \n\
    \n    # 2. Mangle - Connection mark (client side only - NAT friendly)\
    \n    :if ([:len [/ip firewall mangle find where new-connection-mark=\$connMark and comment~\"update_pcq\"]] = 0) do={\
    \n        /ip firewall mangle\
    \n        add chain=prerouting action=mark-connection new-connection-mark=\$connMark \\\
    \n            connection-state=new src-address-list=\$listName \\\
    \n            passthrough=yes comment=(\"update_pcq \" . \$connMark . \" conn\")\
    \n        :log info (\$logPrefix . \"Created connection mark \" . \$connMark)\
    \n    }\
    \n\
    \n    # 3. Mangle - Packet marks\
    \n    :if ([:len [/ip firewall mangle find where new-packet-mark=\$pktDown and comment~\"update_pcq\"]] = 0) do={\
    \n        /ip firewall mangle\
    \n        # Download - connection-mark only (works with NAT)\
    \n        add chain=prerouting action=mark-packet new-packet-mark=\$pktDown \\\
    \n            connection-mark=\$connMark \\\
    \n            passthrough=yes comment=(\"update_pcq \" . \$pktDown)\
    \n\
    \n        # Upload\
    \n        add chain=prerouting action=mark-packet new-packet-mark=\$pktUp \\\
    \n            connection-mark=\$connMark src-address-list=\$listName \\\
    \n            passthrough=yes comment=(\"update_pcq \" . \$pktUp)\
    \n\
    \n        :log info (\$logPrefix . \"Created packet marks for \" . \$connMark)\
    \n    }\
    \n\
    \n    # 4. Simple Queues - SEPARATE UP / DOWN + force target\
    \n    :if ([:len [/queue simple find where name=\$simpleUp]] = 0) do={\
    \n        /queue simple add name=\$simpleUp \\\
    \n            packet-marks=\$pktUp \\\
    \n            queue=\$pcqUp \\\
    \n            max-limit=(\"0/0\") \\\
    \n            target=0.0.0.0/0 \\\
    \n            comment=(\"update_pcq \" . \$listName . \" UP\")\
    \n        :log info (\$logPrefix . \"Created \" . \$simpleUp)\
    \n    } else={\
    \n        # Ensure all properties match on existing queues\
    \n        /queue simple set [find name=\$simpleUp] \\\
    \n            packet-marks=\$pktUp \\\
    \n            queue=\$pcqUp \\\
    \n            max-limit=(\"0/0\") \\\
    \n            target=0.0.0.0/0 \\\
    \n            comment=(\"update_pcq \" . \$listName . \" UP\")\
    \n    }\
    \n\
    \n    :if ([:len [/queue simple find where name=\$simpleDown]] = 0) do={\
    \n        /queue simple add name=\$simpleDown \\\
    \n            packet-marks=\$pktDown \\\
    \n            queue=(\"default-small/\" . \$pcqDown) \\\
    \n            max-limit=(\"0/0\") \\\
    \n            target=0.0.0.0/0 \\\
    \n            comment=(\"update_pcq \" . \$listName . \" DOWN\")\
    \n        :log info (\$logPrefix . \"Created \" . \$simpleDown)\
    \n    } else={\
    \n        # Ensure all properties match on existing queues\
    \n        /queue simple set [find name=\$simpleDown] \\\
    \n            packet-marks=\$pktDown \\\
    \n            queue=(\"default-small/\" . \$pcqDown) \\\
    \n            max-limit=(\"0/0\") \\\
    \n            target=0.0.0.0/0 \\\
    \n            comment=(\"update_pcq \" . \$listName . \" DOWN\")\
    \n    }\
    \n}\
    \n\
    \n# ---------- Cleanup: remove orphaned PCQ infrastructure ----------\
    \n:log info (\$logPrefix . \"Starting cleanup of orphaned packages...\")\
    \n\
    \n:foreach sq in=[/queue simple find where comment~\"update_pcq\"] do={\
    \n    :local sqName [/queue simple get \$sq name]\
    \n\
    \n    # Expected formats: SP-1000-50-UP  or  SP-1000-50-DOWN\
    \n    :if ([:pick \$sqName 0 3] = \"SP-\") do={\
    \n        :local rest [:pick \$sqName 3 [:len \$sqName]]\
    \n\
    \n        # Remove -UP or -DOWN from the end\
    \n        :local cleanName \$rest\
    \n        :if ([:pick \$rest ([:len \$rest] - 3) [:len \$rest]] = \"-UP\") do={\
    \n            :set cleanName [:pick \$rest 0 ([:len \$rest] - 3)]\
    \n        }\
    \n        :if ([:pick \$rest ([:len \$rest] - 5) [:len \$rest]] = \"-DOWN\") do={\
    \n            :set cleanName [:pick \$rest 0 ([:len \$rest] - 5)]\
    \n        }\
    \n\
    \n        # cleanName should now be like \"1000-50\"\
    \n        :local dashPos [:find \$cleanName \"-\"]\
    \n        :if (\$dashPos != nil) do={\
    \n            :local dl [:pick \$cleanName 0 \$dashPos]\
    \n            :local ul [:pick \$cleanName (\$dashPos + 1) [:len \$cleanName]]\
    \n            :local expectedList (\"Speed_Package-\" . \$dl . \"/\" . \$ul)\
    \n\
    \n            :local memberCount [:len [/ip firewall address-list find where list=\$expectedList]]\
    \n\
    \n            :if (\$memberCount = 0) do={\
    \n                :log warning (\$logPrefix . \"Removing orphaned queue \" . \$sqName . \" (list \" . \$expectedList . \" is empty)\")\
    \n\
    \n                :local connMark   (\"sp-\" . \$dl . \"-\" . \$ul)\
    \n                :local pktDown    (\$connMark . \"-down\")\
    \n                :local pktUp      (\$connMark . \"-up\")\
    \n                :local pcqDown    (\"pcq-\" . \$connMark . \"-down\")\
    \n                :local pcqUp      (\"pcq-\" . \$connMark . \"-up\")\
    \n                :local baseName   (\"SP-\" . \$dl . \"-\" . \$ul)\
    \n\
    \n                /queue simple remove \$sq\
    \n\
    \n                # Only clean related objects when BOTH UP and DOWN queues are gone\
    \n                :if ([:len [/queue simple find where name~(\$baseName)]] = 0) do={\
    \n                    /ip firewall mangle remove [find where comment~(\"update_pcq \" . \$connMark)]\
    \n                    /ip firewall mangle remove [find where comment~(\"update_pcq \" . \$pktDown)]\
    \n                    /ip firewall mangle remove [find where comment~(\"update_pcq \" . \$pktUp)]\
    \n\
    \n                    :if ([:len [/queue type find where name=\$pcqDown]] > 0) do={\
    \n                        /queue type remove [find where name=\$pcqDown]\
    \n                    }\
    \n                    :if ([:len [/queue type find where name=\$pcqUp]] > 0) do={\
    \n                        /queue type remove [find where name=\$pcqUp]\
    \n                    }\
    \n                }\
    \n            }\
    \n        }\
    \n    }\
    \n}\
    \n\
    \n:log info (\$logPrefix . \"Finished.\")"
/file/remove [find name=update_pcq.rsc];
