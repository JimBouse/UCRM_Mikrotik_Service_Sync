/system script remove [find name=update_dhcp_leases];
/system script add dont-require-permissions=no name=update_dhcp_leases source="# ============================================\
    \n# Script: update_dhcp_leases\
    \n# Purpose: Remove DHCP leases last-seen older\
    \n#          than dhcpLeaseStaleDays, then convert\
    \n#          remaining dynamic leases to static\
    \n# Globals (set by plugin before run):\
    \n#   dhcpLeaseStaleDays   - number, 0 = skip prune\
    \n#   dhcpLeaseServerName  - DHCP server, \"\" = all\
    \n# ============================================\
    \n\
    \n:global dhcpLeaseStaleDays\
    \n:global dhcpLeaseServerName\
    \n\
    \n:local logPrefix \"update_dhcp_leases: \"\
    \n:local staleDays 7\
    \n:local serverName \"\"\
    \n\
    \n:if ([:typeof \$dhcpLeaseStaleDays] = \"num\") do={\
    \n    :set staleDays \$dhcpLeaseStaleDays\
    \n} else={\
    \n    :if (([:typeof \$dhcpLeaseStaleDays] = \"str\") && ([:len \$dhcpLeaseStaleDays] > 0)) do={\
    \n        :set staleDays [:tonum \$dhcpLeaseStaleDays]\
    \n    }\
    \n}\
    \n\
    \n:if ([:typeof \$dhcpLeaseServerName] = \"str\") do={\
    \n    :set serverName \$dhcpLeaseServerName\
    \n}\
    \n\
    \n:local serverLabel \"*\"\
    \n:if ([:len \$serverName] > 0) do={\
    \n    :set serverLabel \$serverName\
    \n}\
    \n\
    \n:log info (\$logPrefix . \"Starting... server=\" . \$serverLabel . \" staleDays=\" . \$staleDays)\
    \n\
    \n:local leaseIds\
    \n:if ([:len \$serverName] > 0) do={\
    \n    :set leaseIds [/ip dhcp-server lease find where server=\$serverName]\
    \n} else={\
    \n    :set leaseIds [/ip dhcp-server lease find]\
    \n}\
    \n\
    \n:local removed 0\
    \n:local converted 0\
    \n\
    \n:if (\$staleDays > 0) do={\
    \n    :local maxAge [:totime ([:tostr \$staleDays] . \"d\")]\
    \n    :foreach i in=\$leaseIds do={\
    \n        :onerror err in={\
    \n            :local ls [/ip dhcp-server lease get \$i last-seen]\
    \n            :if ([:typeof \$ls] = \"time\") do={\
    \n                :if (\$ls > \$maxAge) do={\
    \n                    /ip dhcp-server lease remove \$i\
    \n                    :set removed (\$removed + 1)\
    \n                }\
    \n            }\
    \n        } do={\
    \n            :log warning (\$logPrefix . \"prune failed: \" . \$err)\
    \n        }\
    \n    }\
    \n}\
    \n\
    \n:if ([:len \$serverName] > 0) do={\
    \n    :set leaseIds [/ip dhcp-server lease find where server=\$serverName dynamic=yes]\
    \n} else={\
    \n    :set leaseIds [/ip dhcp-server lease find where dynamic=yes]\
    \n}\
    \n\
    \n:foreach i in=\$leaseIds do={\
    \n    :onerror err in={\
    \n        /ip dhcp-server lease make-static \$i\
    \n        :set converted (\$converted + 1)\
    \n    } do={\
    \n        :log warning (\$logPrefix . \"make-static failed: \" . \$err)\
    \n    }\
    \n}\
    \n\
    \n:log info (\$logPrefix . \"Finished. converted=\" . \$converted . \" removed=\" . \$removed)"
/file/remove [find name=update_dhcp_leases.rsc];
